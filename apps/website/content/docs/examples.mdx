## 1. High-Security 2FA / OTP with Dynamic Omnichannel Fallback

This recipe dispatches an authentication code via SMS, and automatically falls back to WhatsApp if the SMS provider encounters delivery degradation or carrier timeouts.

<Tabs items={['TypeScript SDK', 'cURL']}>
  <Tab value="TypeScript SDK">
```typescript
import { Convey } from '@convey/sdk';

const convey = new Convey({
  apiKey: process.env.CONVEY_API_KEY!,
  baseUrl: 'https://api.convey.internal',
});

async function sendTwoFactorOtp(userId: string, phoneNumber: string, otpCode: string) {
  const response = await convey.messages.send({
    channel: 'SMS',
    recipient: phoneNumber,
    priority: 'CRITICAL',
    content: {
      body: `Your security code is ${otpCode}. Valid for 10 minutes. Never share this code.`,
    },
    category: 'SECURITY',
    metadata: {
      userId,
      purpose: 'security_2fa_login',
      ipAddress: '198.51.100.42',
    },
    idempotencyKey: `otp_${userId}_${Math.floor(Date.now() / 60000)}`,
  });

  console.log(`Accepted OTP dispatch: ${response.publicId} (Status: ${response.status})`);
  return response;
}
```

## 2. Transactional Invoices with Dynamic Attachments

This recipe dispatches branded transactional invoice emails with attachments and strict delivery tracking.

<Tabs items={['TypeScript SDK', 'cURL']}>
  <Tab value="TypeScript SDK">
```typescript
import { Convey } from '@convey/sdk';

const convey = new Convey({ apiKey: process.env.CONVEY_API_KEY! });

async function sendInvoiceEmail(invoice: {
  id: string;
  customerEmail: string;
  amount: string;
  pdfBase64: string;
}) {
  return await convey.messages.send({
    channel: 'EMAIL',
    recipient: invoice.customerEmail,
    priority: 'DEFAULT',
    content: {
      subject: `Receipt for Invoice #${invoice.id}`,
      body: `Hello, thank you for your business. Your payment of ${invoice.amount} has been processed successfully.`,
      attachments: [
        {
          filename: `invoice_${invoice.id}.pdf`,
          contentType: 'application/pdf',
          content: invoice.pdfBase64,
        },
      ],
    },
    category: 'BILLING',
    idempotencyKey: `inv_${invoice.id}_dispatch`,
  });
}
```

## 3. WhatsApp Interactive Notifications

```typescript
import { Convey } from '@convey/sdk';

const convey = new Convey({ apiKey: process.env.CONVEY_API_KEY! });

async function sendOrderShippingUpdate(phone: string, orderId: string, trackingNumber: string) {
  return await convey.messages.send({
    channel: 'WHATSAPP',
    recipient: phone,
    content: {
      templateId: 'shipping_notification_v2',
      variables: {
        orderId,
        trackingNumber,
      },
    },
    metadata: { orderId },
  });
}
```

## 4. Mass Bulk Dispatches with Transactional Outbox

```typescript
import { Convey } from '@convey/sdk';

const convey = new Convey({ apiKey: process.env.CONVEY_API_KEY! });

async function broadcastMarketingNewsletter(subscribers: string[], campaignId: string) {
  const messages = subscribers.map((email, idx) => ({
    channel: 'EMAIL' as const,
    recipient: email,
    priority: 'LOW' as const,
    content: {
      subject: 'Special Product Announcement — Convey v1.0',
      body: '<p>Check out our new planetary-scale communication features.</p>',
    },
    category: 'MARKETING',
    idempotencyKey: `camp_${campaignId}_sub_${idx}`,
  }));

  // Chunk into 500-message micro-batches for 1-RTT transactional ingestion
  for (let i = 0; i < messages.length; i += 500) {
    const chunk = messages.slice(i, i + 500);
    const result = await convey.messages.sendBulk(chunk);
    console.log(`Accepted chunk ${i / 500 + 1}: ${result.total} messages`);
  }
}
```

## 6. Dead-Letter Queue (DLQ) Mutated Replay

```typescript
import { Convey } from '@convey/sdk';

const convey = new Convey({ apiKey: process.env.CONVEY_API_KEY! });

async function fixAndReplayFailedMessage(messageId: string, correctedEmail: string) {
  // 1. Dry run simulation
  const dryRun = await convey.dlq.replayMutated({
    messageIds: [messageId],
    dryRun: true,
    mutations: {
      recipients: { email: correctedEmail },
    },
  });

  console.log(`Dry run status:`, dryRun.messages);

  // 2. Execute live mutated replay
  const live = await convey.dlq.replayMutated({
    messageIds: [messageId],
    dryRun: false,
    mutations: {
      recipients: { email: correctedEmail },
      metadata: { remediationReason: 'Customer corrected email typo in portal' },
    },
  });

  console.log(`Replayed message: ${live.replayedCount} succeeded.`);
}
```

## 7. Inbound Webhook Cryptographic Verification

```typescript
import { Convey, ConveySecurityError } from '@convey/sdk';

const WEBHOOK_SECRET = process.env.CONVEY_WEBHOOK_SECRET!;

export async function handleProviderDeliveryWebhook(req: Request) {
  const signature = req.headers.get('x-convey-signature') || '';
  const rawBody = await req.text();

  try {
    const event = await Convey.webhooks.constructEvent(rawBody, signature, WEBHOOK_SECRET);
    console.log(`Verified webhook event ${event.id}: ${event.type}`);

    return Response.json({ received: true });
  } catch (err) {
    if (err instanceof ConveySecurityError) {
      return new Response('Unauthorized Signature', { status: 401 });
    }
    return new Response('Internal error', { status: 500 });
  }
}
```
